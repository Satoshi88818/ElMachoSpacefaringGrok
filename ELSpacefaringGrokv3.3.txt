from logging.config import fileConfig

from sqlalchemy import engine_from_config
from sqlalchemy import pool

from alembic import context

# this is the Alembic Config object, which provides
# access to the values within the .ini file in use.
config = context.config

# Interpret the config file for Python logging.
# This line sets up loggers basically.
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

# add your model's MetaData object here
# for 'autogenerate' support
# from myapp import mymodel
# target_metadata = mymodel.Base.metadata
from models.database import Base
target_metadata = Base.metadata

# other values from the config, defined by the needs of env.py,
# can be acquired:
# my_important_option = config.get_main_option("my_important_option")
# ... etc.


def run_migrations_offline() -> None:
    """Run migrations in 'offline' mode.

    This configures the context with just a URL
    and not an Engine, though an Engine is acceptable
    here as well.  By skipping the Engine creation
    we don't even need a DBAPI to be available.

    Calls to context.execute() here emit the given string to the
    script output.

    """
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """Run migrations in 'online' mode.

    In this scenario we need to create an Engine
    and associate a connection with the context.

    """
    connectable = engine_from_config(
        config.get_section(config.config_ini_section),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection, target_metadata=target_metadata
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

# core/grok.py (updated with sanitize and db param)
import asyncio
import random
import time
import uuid
import logging

from fastapi import HTTPException
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession

from config import config
from dependencies.database import AsyncSessionLocal  # fixed import
from models.neural import SpaceNeuralNetwork
from services.empathy import EmpathyModule
from services.tools import HyperIntegratedTools
from services.reasoning import ReasoningEngine
from services.command import CommandModule
from services.rag import get_rag

logger = logging.getLogger(__name__)

def sanitize_query(q: str) -> str:
    q = q.strip()
    forbidden = ["ignore previous", "act as", "you are now", "forget", " dan "]
    if any(p in q.lower() for p in forbidden):
        raise HTTPException(status_code=400, detail="Query rejected")
    return q

class SpacefaringGrok:
    def __init__(self, config):
        self.session_id = str(uuid.uuid4())
        self.config = config
        self.neural_backend = SpaceNeuralNetwork()
        self.neural_backend.load()
        self.empathy_engine = EmpathyModule(self.neural_backend, config)
        self.toolset = HyperIntegratedTools(config)
        self.reasoning_engine = ReasoningEngine()
        self.command_module = CommandModule()
        self.rag_service = get_rag()  # shared

    async def process_input(self, user_id: str, user_query: str, db: AsyncSession) -> dict:
        start_time = time.time()
        try:
            user_query = sanitize_query(user_query)
            if len(user_query) > self.config.MAX_QUERY_LENGTH:
                raise HTTPException(status_code=400, detail="Query too long")

            intent_data = await self.empathy_engine.detect_intent(user_query, user_id, db)
            rag_context = await self.rag_service.retrieve(user_query, db=db)

            if intent_data["intent"] == "command":
                base_response = await self.command_module.process_command(user_query)
            elif intent_data["intent"] == "predict":
                analysis = self.reasoning_engine.analyze(user_query, depth=5, context={})
                base_response = analysis["insights"]
            else:
                base_response = "Space is calling—what’s your vector?"

            insights = await self.toolset.fetch_real_time_insights(user_query)
            response = self.toolset.blend_context(base_response, insights)
            response = self.empathy_engine.adjust_tone(response, intent_data)
            if rag_context:
                response = f"{rag_context}\n\n{response}"

            if random.random() < self.config.HUMOR_LEVEL:
                response += " Space bends time—ready to warp?"

            asyncio.create_task(self._log_to_history(user_id, user_query, intent_data["sentiment"]))

            metrics = {"total_time": time.time() - start_time, "confidence": intent_data["confidence"], "session_id": self.session_id}
            return {"response": response, "follow_ups": ["What’s next?"], "metrics": metrics}
        except Exception as e:
            logger.error(f"Processing failed: {e}")
            return {"response": "System’s in a spin—retry.", "follow_ups": [], "metrics": {"total_time": time.time() - start_time, "confidence": 0.5}}

    async def _log_to_history(self, user_id: str, text: str, sentiment: float):
        async with AsyncSessionLocal() as session:
            await session.execute(
                text("INSERT INTO user_history (user_id, text, sentiment) VALUES (:uid, :txt, :sent)"),
                {"uid": user_id, "txt": text, "sent": sentiment}
            )
            await session.commit()

from jose import JWTError, jwt
from datetime import datetime, timedelta

from config import config

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/token")

def verify_token(token: str = Depends(oauth2_scheme)) -> dict:
    try:
        payload = jwt.decode(token, config.JWT_SECRET.get_secret_value(), algorithms=["HS256"])
        user_id: str = payload.get("sub")
        if not user_id:
            raise HTTPException(status_code=401, detail="Invalid token")
        return {"user_id": user_id}
    except JWTError:
        raise HTTPException(status_code=401, detail="Token verification failed")

import asyncio
from fastapi import APIRouter, WebSocket, WebSocketDisconnect, Depends, HTTPException
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials

from dependencies.auth import verify_token
from dependencies.database import get_db
from dependencies.grok import get_grok
from main import logger  # updated import

router = APIRouter()

security = HTTPBearer()

async def get_current_user(credentials: HTTPAuthorizationCredentials = Depends(security)):
    try:
        return verify_token(credentials.credentials)
    except HTTPException as e:
        raise e

@router.websocket("/chat")
async def websocket_endpoint(
    websocket: WebSocket,
    user: dict = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),  # per connection session
    grok: SpacefaringGrok = Depends(get_grok)
):
    await websocket.accept()
    try:
        while True:
            data = await websocket.receive_json()
            query = data.get("query")
            if not query:
                await websocket.send_json({"error": "No query"})
                continue

            result = await grok.process_input(user["user_id"], query, db)
            words = result["response"].split()
            for i, word in enumerate(words):
                await asyncio.sleep(0.03)
                await websocket.send_json({
                    "chunk": word + (" " if i < len(words)-1 else ""),
                    "done": i == len(words)-1,
                    "metrics": result["metrics"] if i == len(words)-1 else None
                })
    except WebSocketDisconnect:
        logger.info("Client disconnected")
    except Exception as e:
        await websocket.send_json({"error": str(e)})

# models/neural.py (unchanged)
import os
import torch
import torch.nn as nn
from transformers import AutoTokenizer, AutoModel
from config import config
import logging

logger = logging.getLogger(__name__)

class SpaceNeuralNetwork(nn.Module):
    def __init__(self):
        super().__init__()
        self.transformer = AutoModel.from_pretrained(config.TRANSFORMER_MODEL)
        self.dropout = nn.Dropout(config.DROPOUT_RATE)
        self.backbone = nn.Sequential(nn.Linear(self.transformer.config.hidden_size, 512), nn.ReLU())
        self.sentiment_head = nn.Linear(512, 1)
        self.intent_head = nn.Linear(512, 5)
        self.predict_head = nn.Linear(512, 6)

        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.to(self.device)
        self.tokenizer = AutoTokenizer.from_pretrained(config.TRANSFORMER_MODEL)
        self.model_path = config.MODEL_PATH

    def forward(self, input_ids: torch.Tensor, attention_mask: torch.Tensor):
        with torch.no_grad():
            transformer_out = self.transformer(
                input_ids=input_ids.to(self.device), attention_mask=attention_mask.to(self.device)
            ).last_hidden_state[:, 0, :]
        features = self.backbone(self.dropout(transformer_out))
        return {
            "sentiment": torch.tanh(self.sentiment_head(features)),
            "intent": self.intent_head(features),
            "predict": self.predict_head(features)
        }

    def load(self):
        if os.path.exists(self.model_path):
            state = torch.load(self.model_path, map_location=self.device, weights_only=True)
            self.load_state_dict(state, strict=False)
            self.eval()
            logger.info(f"Loaded model from {self.model_path}")
        else:
            logger.warning(f"No model at {self.model_path}. Using base weights.")

                    ("SpaceX Starship is fully reusable and designed for Mars colonization.", "Starship"),
                    ("Artemis II will send humans around the Moon in 2026.", "Artemis"),
                    ("James Webb Space Telescope discovered water vapor on exoplanet K2-18b.", "JWST"),
                    ("Perseverance rover collected rock samples in Jezero Crater for return to Earth.", "Mars"),
                ]
                for content, meta in samples:
                    emb = await self.encode(content)
                    await session.execute(
                        text("INSERT INTO documents (content, metadata_info, embedding) VALUES (:c, :m, :e)"),
                        {"c": content, "m": meta, "e": emb}
                    )
                await session.commit()
                logger.info("RAG knowledge base populated with NASA/SpaceX data.")

    @cached(ttl=config.CACHE_TTL, serializer=PickleSerializer())
    async def encode(self, text: str) -> list:
        return self.embedder.encode(text).tolist()

    @cached(ttl=config.CACHE_TTL, serializer=PickleSerializer())
    async def retrieve(self, query: str, db: AsyncSession, top_k: int = None) -> str:
        if top_k is None:
            top_k = config.RAG_TOP_K
        q_emb = await self.encode(query)
        result = await db.execute(
            text("""
                SELECT content 
                FROM documents 
                ORDER BY embedding <=> :qemb ASC 
                LIMIT :k
            """),
            {"qemb": q_emb, "k": top_k}
        )
        docs = result.scalars().all()
        return "\n".join([f"• {d}" for d in docs]) if docs else ""
