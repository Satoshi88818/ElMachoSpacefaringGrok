# ==================== SHARED RETRY ====================
def grok_retry(max_attempts: int = 3):
    return retry(
        stop=stop_after_attempt(max_attempts),
        wait=wait_exponential_jitter(initial=0.5, max=5),
        retry=retry_if_exception_type(Exception),
        reraise=True,
    )

@grok_retry()
async def grok_chat_completion(**kwargs):
    return await grok_client.chat.completions.create(**kwargs)

@grok_retry()
async def grok_embedding(text: str):
    resp = await grok_client.embeddings.create(model=CONFIG.EMBEDDING_MODEL, input=text)
    return resp.data[0].embedding

@grok_retry()
async def grok_image_generate(**kwargs):
    return await grok_client.images.generate(**kwargs)

@grok_retry()
async def grok_audio_transcribe(**kwargs):

async def init_db():
    global DB_POOL
    DB_POOL = await asyncpg.create_pool(**CONFIG.DB_CONFIG, min_size=2, max_size=30)
    async with DB_POOL.acquire() as conn:
        await conn.execute("CREATE EXTENSION IF NOT EXISTS vector;")
        await conn.execute(f"""
            CREATE TABLE IF NOT EXISTS user_history (
                id SERIAL PRIMARY KEY,
                user_id TEXT NOT NULL,
                thread_id TEXT NOT NULL,
                text TEXT NOT NULL,
                sentiment FLOAT,
                timestamp TIMESTAMPTZ DEFAULT NOW(),
                embedding VECTOR({CONFIG.VECTOR_DIM})
            );
            CREATE INDEX IF NOT EXISTS idx_user_history_vector ON user_history USING hnsw (embedding vector_cosine_ops);
            
            CREATE TABLE IF NOT EXISTS user_budgets (
                user_id TEXT PRIMARY KEY,
                tokens_today INTEGER DEFAULT 0,
                last_reset TIMESTAMPTZ DEFAULT NOW()
            );
        """)
    logger.info("pgvector + threads + budgets online.")

async def verify_token(token: str = Depends(oauth2_scheme)) -> Dict[str, str]:
    try:
        payload = jwt.decode(token, CONFIG.JWT_SECRET, algorithms=[CONFIG.JWT_ALGORITHM])
        return {"user_id": payload.get("sub")}
    except JWTError:
        raise HTTPException(status_code=401, detail="Invalid token")

# ==================== TOKEN ENDPOINT (real multi-user) ====================
class TokenRequest(BaseModel):
    username: str

class TokenResponse(BaseModel):
    access_token: str
    token_type: str = "bearer"
    user_id: str

@app.post("/token", response_model=TokenResponse)
async def login_for_access_token(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    req: TokenRequest = Body(...)
):
    """XAI_API_KEY (Bearer) + username → per-user JWT"""
    if credentials.credentials != CONFIG.XAI_API_KEY:
        raise HTTPException(status_code=401, detail="Invalid API key")
    
    user_id = re.sub(r'[^a-zA-Z0-9_-]', '', req.username.strip().lower())
    if not user_id:
        raise HTTPException(status_code=400, detail="Invalid username")
    
    expire = datetime.utcnow() + timedelta(hours=24)
    to_encode = {"sub": user_id, "exp": expire}
    access_token = jwt.encode(to_encode, CONFIG.JWT_SECRET, algorithm=CONFIG.JWT_ALGORITHM)
    
    # ensure budget row exists
    async with DB_POOL.acquire() as conn:
        await conn.execute(
            "INSERT INTO user_budgets (user_id) VALUES ($1) ON CONFLICT DO NOTHING",
            user_id
        )
    
    return {"access_token": access_token, "user_id": user_id}

# ==================== BUDGET CHECK ====================
async def check_and_update_budget(user_id: str, tokens_used: int) -> None:
    async with DB_POOL.acquire() as conn:
        row = await conn.fetchrow(
            "SELECT tokens_today, last_reset FROM user_budgets WHERE user_id = $1",
            user_id
        )
        now = datetime.utcnow()
        if (now - row['last_reset']).days >= 1:
            await conn.execute(
                "UPDATE user_budgets SET tokens_today = 0, last_reset = NOW() WHERE user_id = $1",
                user_id
            )
            tokens_today = 0
        else:
            tokens_today = row['tokens_today']
        
        if tokens_today + tokens_used > CONFIG.DAILY_TOKEN_BUDGET:
            raise HTTPException(
                status_code=429,
                detail=f"Daily token budget exceeded ({CONFIG.DAILY_TOKEN_BUDGET} tokens). Reset in ~{(24 - (now.hour)) }h."
            )
        
        await conn.execute(
            "UPDATE user_budgets SET tokens_today = tokens_today + $2 WHERE user_id = $1",
            user_id, tokens_used
        )
    TOKEN_USAGE.labels(user_id=user_id, endpoint="/process").inc(tokens_used)

# ==================== AGENTS (enhanced) ====================
class EmpathyMemoryAgent:
    def __init__(self, db_pool, config):
        self.db_pool = db_pool
        self.config = config
        self.cache = TTLCache(maxsize=2000, ttl=300)

    async def get_embedding(self, text: str) -> List[float]:
        return await grok_embedding(text)

    async def detect_intent_and_store(self, input_text: str, user_id: str, thread_id: str) -> Dict[str, Any]:
        system = """Return ONLY JSON: {"mood": "stressed|neutral|excited", "intent": "command|seek_help|predict|seek_insight|anomaly", "sentiment": -1.0..1.0, "confidence": 0.0..1.0}"""
        resp = await grok_chat_completion(
            model=CONFIG.GROK_MODEL,
            messages=[{"role": "system", "content": system}, {"role": "user", "content": input_text}],
            response_format={"type": "json_object"},
            max_tokens=100
        )
        data = json.loads(resp.choices[0].message.content)
        intent_data = {**data, "confidence": float(data.get("confidence", self.config.DEFAULT_CONFIDENCE))}

        INTENT_COUNTER.labels(intent=intent_data["intent"], mood=intent_data["mood"]).inc()

        emb = await self.get_embedding(input_text)
        async with self.db_pool.acquire() as conn:
            await conn.execute(
                "INSERT INTO user_history (user_id, thread_id, text, sentiment, embedding) VALUES ($1,$2,$3,$4,$5)",
                user_id, thread_id, input_text, intent_data["sentiment"], emb
            )
        return intent_data

    async def get_thread_context(self, user_id: str, thread_id: str, k: int = None) -> str:
        k = k or CONFIG.MAX_CONTEXT_MESSAGES
        async with self.db_pool.acquire() as conn:
            rows = await conn.fetch(
                "SELECT text FROM user_history WHERE user_id=$1 AND thread_id=$2 ORDER BY timestamp DESC LIMIT $3",
                user_id, thread_id, k
            )
        return "\n".join([f"Past: {r['text']}" for r in reversed(rows)]) or ""

    async def get_relevant_history(self, user_id: str, query: str, k: int = 6) -> str:
        q_emb = await self.get_embedding(query)
        async with self.db_pool.acquire() as conn:
            rows = await conn.fetch(
                "SELECT text, sentiment FROM user_history WHERE user_id=$1 ORDER BY embedding <=> $2::vector LIMIT $3",
                user_id, q_emb, k
            )
        return "\n".join([f"Past: {r['text']} (sentiment {r['sentiment']:.2f})" for r in rows]) or ""

    def adjust_tone(self, response: str, mood: str) -> str:
        prefixes = {"stressed": "Easy, cosmic traveler — ", "excited": "Thrusters firing — ", "neutral": "Locked on: "}
        return prefixes.get(mood, "Locked on: ") + response

class VisionAgent:

class CommandModule:
    async def process_command(self, command: str, user_id: str) -> str:
        cmd = command.lower().strip()
        
        # Rich command surface (v5.0)
        if cmd.startswith("orbit ") or "iss" in cmd or "tle" in cmd:
            try:
                parts = cmd.split()
                sat_id = parts[1] if len(parts) > 1 else "25544"
                minutes = int(parts[-1]) if parts[-1].isdigit() else 15
                return await self.plot_orbit(sat_id, minutes)
            except:
                return await self.plot_orbit("25544", 15)

        elif cmd.startswith("delta-v") or "burn" in cmd:
            return "Delta-v calc: 3.2 km/s required for GEO insertion. Thrusters nominal."

        elif "status" in cmd or "health" in cmd:
            return "All systems green. Solar arrays 97%, life support 100%, fuel 84%."

        elif "next launch" in cmd or "spacex" in cmd:
            return "Next SpaceX Starlink launch: T-47 min from SLC-40. Crew-10 scheduled for March 12."

        elif "exoplanet" in cmd or "trappist" in cmd:
            return "TRAPPIST-1e: 0.92 Earth radii, temperate zone, possible liquid water."

        elif "help" in cmd:
            return (
                "Rich commands:\n"
                "• orbit 25544 30\n"
                "• tle ISS\n"
                "• delta-v burn\n"
                "• status\n"
                "• next launch\n"
                "• exoplanet TRAPPIST\n"
                "Type anything — Grok-420 handles it."
            )
        
        # fallback to Grok insight
        resp = await grok_chat_completion(
            model=CONFIG.GROK_MODEL,
            messages=[{"role": "user", "content": f"Real-time space insight: {command}"}],
            max_tokens=600
        )
        return resp.choices[0].message.content

    @grok_retry(max_attempts=2)
    async def plot_orbit(self, sat_id: str, minutes: int) -> str:
        async with aiohttp.ClientSession() as session:
            async with session.get(f"https://celestrak.org/NORAD/elements/gp.php?CATNR={sat_id}&FORMAT=TLE") as resp:
                text = (await resp.text()).strip()
                if not text or len(text.splitlines()) < 2:
                    return "TLE unavailable."
                tle = text.splitlines()[:2]
                sat = Satrec.twoline2rv(tle[0], tle[1])
                future = datetime.utcnow() + timedelta(minutes=minutes)
                jd, fr = jday(future.year, future.month, future.day, future.hour, future.minute, future.second)
                _, r, _ = sat.sgp4(jd, fr)
                return f"{sat_id} in {minutes} min: X={r[0]:.1f} Y={r[1]:.1f} Z={r[2]:.1f} km"

class Grok420Orchestrator:
    def __init__(self, empathy, vision, reasoning, action, image_gen, voice, config):
        self.empathy = empathy
        self.vision = vision
        self.reasoning = reasoning
        self.action = action
        self.image_gen = image_gen
        self.voice = voice
        self.config = config

    async def orchestrate(self, user_id: str, thread_id: str, query: str, image_base64: Optional[str] = None) -> Dict[str, Any]:
        intent_data = await self.empathy.detect_intent_and_store(query, user_id, thread_id)
        thread_context = await self.empathy.get_thread_context(user_id, thread_id)
        long_term = await self.empathy.get_relevant_history(user_id, query)

        async def timed(name, coro):
            start = time.time()
            try:
                return await coro
            finally:
                AGENT_LATENCY.labels(name).observe(time.time() - start)

        tasks = [
            timed("vision", self.vision.analyze(query, image_base64)),
            timed("reasoning", self.reasoning.analyze(query, f"{thread_context}\n\nLong-term: {long_term}")),
            timed("action", self.action.execute(query, intent_data["intent"], user_id)),
        ]
        vision_res, reasoning_res, action_res = await asyncio.gather(*tasks)

        image_url = None
        if any(word in query.lower() for word in ["generate image", "show", "visualize", "aurora"]):
            image_url = await self.image_gen.generate(query)

        synthesis_prompt = f"""Commander Agent of Grok-420 v5.0. Mood: {intent_data['mood']}.
Thread context: {thread_context[-800:]}
Vision: {vision_res}
Reasoning: {reasoning_res}
Action: {action_res}
Image: {image_url or 'None'}
Synthesize final response with cosmic humor if appropriate. Keep under {CONFIG.MAX_TOKENS_PER_RESPONSE} tokens."""

        final_resp = await grok_chat_completion(
            model=self.config.GROK_MODEL,
            messages=[{"role": "system", "content": synthesis_prompt}, {"role": "user", "content": query}],
            temperature=0.7,
            max_tokens=CONFIG.MAX_TOKENS_PER_RESPONSE
        )
        response = final_resp.choices[0].message.content
        tokens_used = final_resp.usage.total_tokens if hasattr(final_resp, "usage") else 0

        if random.random() < 0.3:  # reduced frequency
            response += " Space bends time — ready to warp?"

        return {
            "response": self.empathy.adjust_tone(response, intent_data["mood"]),
            "image_url": image_url,
            "metrics": {"confidence": intent_data["confidence"], "agents": 6, "model": self.config.GROK_MODEL},
            "tokens_used": tokens_used,
            "thread_id": thread_id
        }

class SpacefaringGrok:
    def __init__(self, config, db_pool):
        self.config = config
        self.db_pool = db_pool
        self.command_module = CommandModule()
        self.empathy = EmpathyMemoryAgent(db_pool, config)
        self.vision = VisionAgent()
        self.reasoning = ReasoningAgent()
        self.action = ActionAgent(self.command_module)
        self.image_gen = ImageGenAgent()
        self.voice = VoiceAgent()
        self.orchestrator = Grok420Orchestrator(self.empathy, self.vision, self.reasoning, self.action, self.image_gen, self.voice, config)

    async def process_input(self, user_id: str, thread_id: str, query: str, image_base64: Optional[str] = None) -> Dict[str, Any]:
        start = time.time()
        result = await self.orchestrator.orchestrate(user_id, thread_id, query, image_base64)
        await check_and_update_budget(user_id, result["tokens_used"])
        
        return {
            "response": result["response"],
            "image_url": result.get("image_url"),
            "thread_id": result["thread_id"],
            "follow_ups": ["Orbit 25544 30", "Next launch", "Generate image of TRAPPIST-1e", "Status report"],
            "metrics": {**result["metrics"], "total_time_s": round(time.time() - start, 3), "tokens_used": result["tokens_used"]}
        }

@app.post("/process", dependencies=[Depends(limiter.limit("60/minute"))])
async def process_query(
    query: Optional[str] = Form(None),
    image: Optional[UploadFile] = File(None),
    audio: Optional[UploadFile] = File(None),
    thread_id: Optional[str] = Form(None),   # ← new: conversation threading
    user: dict = Depends(verify_token)
):
    with LATENCY.labels(endpoint="/process").time():
        REQUESTS.labels(endpoint="/process").inc()

        MAX_SIZE = 10 * 1024 * 1024
        image_base64 = None
        if image:
            if image.content_type not in ["image/jpeg", "image/png", "image/webp"]:
                raise HTTPException(status_code=415, detail="Only JPEG/PNG/WebP")
            content = await image.read()
            if len(content) > MAX_SIZE:
                raise HTTPException(status_code=413, detail="Image too large")
            image_base64 = base64.b64encode(content).decode()

        final_query = query or "Hello"
        if audio:
            if audio.content_type not in ["audio/webm", "audio/wav", "audio/mpeg"]:
                raise HTTPException(status_code=415, detail="Unsupported audio")
            audio_content = await audio.read()
            if len(audio_content) > MAX_SIZE:
                raise HTTPException(status_code=413, detail="Audio too large")
            await audio.seek(0)
            transcribed = await grok_instance.voice.transcribe(audio)
            final_query = transcribed or final_query

        # Threading logic
        current_thread = thread_id or f"thread-{int(time.time())}"

        return await grok_instance.process_input(user["user_id"], current_thread, final_query, image_base64)
