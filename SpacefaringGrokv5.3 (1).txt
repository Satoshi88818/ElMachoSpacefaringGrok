core/circuit_breaker.py

""" core/circuit_breaker.py — Async-safe circuit breaker. FIX (v5.1 → v5.2): The original implementation mutated `failure_count` and `last_failure_time` without any locking. Under asyncio concurrency two coroutines could race on these values. We now guard all state mutations with `asyncio.Lock`. """ import asyncio import time import logging from typing import Any, Callable logger = logging.getLogger(__name__) class CircuitOpenError(Exception): """Raised when the circuit is open and requests are being rejected.""" class AsyncCircuitBreaker: def __init__(self, failure_threshold: int = 3, reset_timeout: int = 300): self._failure_count: int = 0 self._last_failure_time: float = 0.0 self._failure_threshold = failure_threshold self._reset_timeout = reset_timeout self._lock = asyncio.Lock() @property def is_open(self) -> bool: """Non-locking read — safe for a quick pre-check (stale by design).""" return ( self._failure_count >= self._failure_threshold and time.monotonic() - self._last_failure_time < self._reset_timeout ) async def execute(self, func: Callable, *args: Any, **kwargs: Any) -> Any: async with self._lock: now = time.monotonic() if ( self._failure_count >= self._failure_threshold and now - self._last_failure_time < self._reset_timeout ): raise CircuitOpenError( "Circuit open: too many upstream failures. " "Service temporarily unavailable; please retry later." ) try: result = await func(*args, **kwargs) async with self._lock: if self._failure_count > 0: logger.info("Circuit breaker: resetting failure count after success.") self._failure_count = 0 return result except CircuitOpenError: raise except Exception: async with self._lock: self._failure_count += 1 self._last_failure_time = time.monotonic() logger.warning( "Circuit breaker: failure %d/%d recorded.", self._failure_count, self._failure_threshold, ) raise # Module-level singleton — all Grok API calls share one breaker. circuit_breaker = AsyncCircuitBreaker() def with_circuit_breaker(func: Callable) -> Callable: """Decorator that routes a coroutine through the shared circuit breaker.""" async def wrapper(*args: Any, **kwargs: Any) -> Any: return await circuit_breaker.execute(func, *args, **kwargs) wrapper.__name__ = func.__name__ return wrapper 

core/grok_client.py

""" core/grok_client.py — Shared Grok API client, retry logic, and helpers. All agents import `grok_chat_with_usage`, `grok_embedding`, etc. from here. The decorator order is intentional: @with_circuit_breaker wraps the outer function. @grok_retry wraps the inner call. ⟹ retries happen *inside* the circuit-breaker check, which is correct: each retry counts as one "attempt" from the breaker's perspective only if all retries are exhausted and an exception propagates outward. """ from typing import Tuple from openai import AsyncOpenAI from tenacity import ( retry, stop_after_attempt, wait_exponential_jitter, retry_if_exception_type, ) from config import CONFIG from core.circuit_breaker import with_circuit_breaker # Single shared async client. _grok_client = AsyncOpenAI(api_key=CONFIG.XAI_API_KEY, base_url="https://api.x.ai/v1") def grok_retry(max_attempts: int = 3): """Factory for a tenacity retry decorator with jittered exponential back-off.""" return retry( stop=stop_after_attempt(max_attempts), wait=wait_exponential_jitter(initial=0.5, max=5), retry=retry_if_exception_type(Exception), reraise=True, ) # ── Raw API wrappers ────────────────────────────────────────────────────────── @with_circuit_breaker @grok_retry() async def _chat_completion(**kwargs): return await _grok_client.chat.completions.create(**kwargs) @with_circuit_breaker @grok_retry() async def _embedding(text: str): return await _grok_client.embeddings.create( model=CONFIG.EMBEDDING_MODEL, input=text ) @with_circuit_breaker @grok_retry() async def _image_generate(**kwargs): return await _grok_client.images.generate(**kwargs) @with_circuit_breaker @grok_retry() async def _audio_transcribe(**kwargs): return await _grok_client.audio.transcriptions.create(**kwargs) # ── Public helpers ───────────────────────────────────────────────────────────── async def grok_chat_with_usage(**kwargs) -> Tuple[str, int]: """ Call the chat completion endpoint and return (content, total_tokens). Used everywhere so that token accounting is never missed. """ resp = await _chat_completion(**kwargs) content: str = resp.choices[0].message.content tokens: int = ( getattr(resp.usage, "total_tokens", 0) if hasattr(resp, "usage") else 0 ) return content, tokens async def grok_embedding(text: str): resp = await _embedding(text) return resp.data[0].embedding async def grok_image_generate(**kwargs): return await _image_generate(**kwargs) async def grok_audio_transcribe(**kwargs): return await _audio_transcribe(**kwargs)

core/rate_limit.py

""" core/rate_limit.py — Per-user rate limiting via slowapi. The key function extracts the user_id from the JWT so that rate limits are applied per authenticated user rather than per IP (which can be spoofed or shared behind a NAT). Falls back to IP if the token is absent/invalid. """ from fastapi import Request from jose import jwt, JWTError from slowapi import Limiter from slowapi.util import get_remote_address from config import CONFIG def _get_user_rate_limit_key(request: Request) -> str: """ Extract user_id directly from the Authorization header. This runs before Depends(verify_token) so we decode manually. """ auth = request.headers.get("authorization", "") if auth.lower().startswith("bearer "): try: token = auth.split(" ", 1)[1] payload = jwt.decode( token, CONFIG.JWT_SECRET, algorithms=[CONFIG.JWT_ALGORITHM] ) user_id = payload.get("sub", "") if user_id: return f"user:{user_id}" except (JWTError, Exception): pass return get_remote_address(request) limiter = Limiter(key_func=_get_user_rate_limit_key)

db/pool.py

""" db/pool.py — asyncpg connection pool lifecycle and FastAPI dependency. FIX (v5.1 → v5.2): The original code stored the pool in a bare module global `DB_POOL` and accessed it directly inside endpoint handlers and agents. We now store it on `app.state` (set during startup) and expose it via a FastAPI dependency (`get_db_pool`). Agents that need the pool receive it via constructor injection at startup time — they never reach into globals themselves. """ import logging import asyncpg from fastapi import Request from config import CONFIG logger = logging.getLogger(__name__) async def create_pool() -> asyncpg.Pool: """Create the asyncpg pool and run schema migrations.""" pool: asyncpg.Pool = await asyncpg.create_pool( **CONFIG.DB_CONFIG, min_size=CONFIG.DB_MIN_SIZE, max_size=CONFIG.DB_MAX_SIZE, ) async with pool.acquire() as conn: await conn.execute("CREATE EXTENSION IF NOT EXISTS vector;") await conn.execute( f""" CREATE TABLE IF NOT EXISTS user_history ( id SERIAL PRIMARY KEY, user_id TEXT NOT NULL, thread_id TEXT NOT NULL, text TEXT NOT NULL, sentiment FLOAT, timestamp TIMESTAMPTZ DEFAULT NOW(), embedding VECTOR({CONFIG.VECTOR_DIM}) ); CREATE INDEX IF NOT EXISTS idx_user_history_vector ON user_history USING hnsw (embedding vector_cosine_ops); CREATE TABLE IF NOT EXISTS user_budgets ( user_id TEXT PRIMARY KEY, tokens_today INTEGER DEFAULT 0, last_reset TIMESTAMPTZ DEFAULT NOW() ); CREATE TABLE IF NOT EXISTS users ( username TEXT PRIMARY KEY, hashed_password TEXT NOT NULL ); """ ) logger.info("Database pool ready — pgvector + budgets + users online.") return pool async def close_pool(pool: asyncpg.Pool) -> None: await pool.close() logger.info("Database pool closed.") # ── FastAPI dependency ──────────────────────────────────────────────────────── def get_db_pool(request: Request) -> asyncpg.Pool: """ FastAPI dependency that retrieves the pool from app.state. Raises a clean 503 if the pool is not yet initialised. """ pool: asyncpg.Pool | None = getattr(request.app.state, "db_pool", None) if pool is None: from fastapi import HTTPException raise HTTPException(status_code=503, detail="Database not available.") return pool 

db/budget.py

""" db/budget.py — Per-user daily token budget enforcement. FIX (v5.1 → v5.2): The original `check_and_update_budget` was called *after* the LLM had already spent tokens. A user near their limit could always squeeze one extra request through regardless of cost. We now split the logic into two phases: 1. `preflight_budget_check(user_id, estimated_tokens, pool)` — called BEFORE the LLM pipeline starts. Uses an estimated upper-bound token count so we can reject over-budget users immediately. 2. `commit_token_usage(user_id, actual_tokens, pool)` — called AFTER the pipeline completes to record the real token spend. This also fires the Prometheus counter. Both operations use `SELECT … FOR UPDATE` inside a transaction to avoid race conditions under concurrent requests from the same user. """ import logging from datetime import datetime import asyncpg from fastapi import HTTPException from config import CONFIG from core.metrics import metrics logger = logging.getLogger(__name__) # Rough upper-bound estimate used for the pre-flight check. # Covers ~4 agent calls × MAX_TOKENS_PER_RESPONSE + intent/embedding overhead. ESTIMATED_MAX_TOKENS_PER_REQUEST: int = CONFIG.MAX_TOKENS_PER_RESPONSE * 5 async def _get_or_create_budget_row(conn: asyncpg.Connection, user_id: str): """Return the budget row, creating it if absent (within a transaction).""" row = await conn.fetchrow( "SELECT tokens_today, last_reset FROM user_budgets WHERE user_id = $1 FOR UPDATE", user_id, ) if row is None: await conn.execute( "INSERT INTO user_budgets (user_id, tokens_today, last_reset) VALUES ($1, 0, NOW())" " ON CONFLICT DO NOTHING", user_id, ) row = await conn.fetchrow( "SELECT tokens_today, last_reset FROM user_budgets WHERE user_id = $1 FOR UPDATE", user_id, ) return row async def _resolve_current_tokens( conn: asyncpg.Connection, user_id: str, row ) -> int: """ Return the current token count, resetting the counter if a new day has started. Mutates the DB row if a reset is needed. """ now = datetime.utcnow() if (now - row["last_reset"]).days >= 1: await conn.execute( "UPDATE user_budgets SET tokens_today = 0, last_reset = NOW() WHERE user_id = $1", user_id, ) return 0 return int(row["tokens_today"]) async def preflight_budget_check( user_id: str, pool: asyncpg.Pool, estimated_tokens: int = ESTIMATED_MAX_TOKENS_PER_REQUEST ) -> None: """ Raise HTTP 429 if the user would exceed their daily budget. Call this BEFORE starting the LLM pipeline. """ async with pool.acquire() as conn: async with conn.transaction(): row = await _get_or_create_budget_row(conn, user_id) current = await _resolve_current_tokens(conn, user_id, row) if current + estimated_tokens > CONFIG.DAILY_TOKEN_BUDGET: raise HTTPException( status_code=429, detail=( f"Daily token budget would be exceeded. " f"Used: {current}, budget: {CONFIG.DAILY_TOKEN_BUDGET}." ), ) async def commit_token_usage( user_id: str, actual_tokens: int, pool: asyncpg.Pool ) -> None: """ Persist the actual token spend after a successful pipeline run. Also performs a final over-budget guard (belt-and-suspenders). """ async with pool.acquire() as conn: async with conn.transaction(): row = await _get_or_create_budget_row(conn, user_id) current = await _resolve_current_tokens(conn, user_id, row) new_total = current + actual_tokens if new_total > CONFIG.DAILY_TOKEN_BUDGET: logger.warning( "User %s exceeded daily budget post-hoc: %d > %d.", user_id, new_total, CONFIG.DAILY_TOKEN_BUDGET, ) # Still record — the pre-flight should have caught this. new_total = CONFIG.DAILY_TOKEN_BUDGET await conn.execute( "UPDATE user_budgets SET tokens_today = $1 WHERE user_id = $2", new_total, user_id, ) metrics.token_usage.labels(user_id=user_id, endpoint="/process").inc(actual_tokens) logger.debug("User %s: committed %d tokens (new total ~%d).", user_id, actual_tokens, new_total)

agents/empathy.py

""" agents/empathy.py — EmpathyMemoryAgent Responsibilities: • Detect intent and mood from user input (via Grok JSON mode). • Store each message with its embedding in the vector DB. • Retrieve thread context (recent N messages) and long-term semantic memory. • Adjust the final response tone based on detected mood. FIX (v5.1 → v5.2): `adjust_tone` now checks whether the prefix is already present to prevent double-prefixing if the method is ever called more than once on the same string. """ import json import logging from typing import Any, Dict, List, Optional, Tuple import asyncpg from cachetools import TTLCache from config import CONFIG from core.grok_client import grok_chat_with_usage, grok_embedding from core.metrics import metrics logger = logging.getLogger(__name__) _TONE_PREFIXES: Dict[str, str] = { "stressed": "Easy, cosmic traveler — ", "excited": "Thrusters firing — ", "neutral": "Locked on: ", } _INTENT_SYSTEM_PROMPT = ( 'Return ONLY valid JSON (no markdown, no commentary): ' '{"mood": "stressed|neutral|excited", ' '"intent": "command|seek_help|predict|seek_insight|anomaly", ' '"sentiment": <float -1.0..1.0>, ' '"confidence": <float 0.0..1.0>}' ) class EmpathyMemoryAgent: def __init__(self, db_pool: asyncpg.Pool, config=CONFIG): self._pool = db_pool self._config = config # Cache embeddings for 5 minutes to avoid re-embedding identical strings. self._embedding_cache: TTLCache = TTLCache(maxsize=2_000, ttl=300) # ── Embeddings ──────────────────────────────────────────────────────────── async def get_embedding(self, text: str) -> List[float]: cached = self._embedding_cache.get(text) if cached is not None: return cached emb = await grok_embedding(text) self._embedding_cache[text] = emb return emb # ── Intent detection + storage ──────────────────────────────────────────── async def detect_intent_and_store( self, input_text: str, user_id: str, thread_id: str ) -> Tuple[Dict[str, Any], int]: content, tokens = await grok_chat_with_usage( model=self._config.GROK_MODEL, messages=[ {"role": "system", "content": _INTENT_SYSTEM_PROMPT}, {"role": "user", "content": input_text}, ], response_format={"type": "json_object"}, max_tokens=100, ) try: data = json.loads(content) except json.JSONDecodeError: logger.warning("Intent JSON parse failed; using defaults. Raw: %s", content) data = {} intent_data: Dict[str, Any] = { "mood": data.get("mood", "neutral"), "intent": data.get("intent", "seek_insight"), "sentiment": float(data.get("sentiment", 0.0)), "confidence": float(data.get("confidence", self._config.DEFAULT_CONFIDENCE)), } metrics.intents.labels( intent=intent_data["intent"], mood=intent_data["mood"] ).inc() # Persist to DB asynchronously (fire-and-forget style — errors are logged, not raised). try: emb = await self.get_embedding(input_text) async with self._pool.acquire() as conn: await conn.execute( "INSERT INTO user_history (user_id, thread_id, text, sentiment, embedding) " "VALUES ($1, $2, $3, $4, $5)", user_id, thread_id, input_text, intent_data["sentiment"], emb, ) except Exception as exc: logger.error("Failed to store user history for %s: %s", user_id, exc) return intent_data, tokens # ── Memory retrieval ────────────────────────────────────────────────────── async def get_thread_context( self, user_id: str, thread_id: str, k: Optional[int] = None ) -> str: k = k or self._config.MAX_CONTEXT_MESSAGES async with self._pool.acquire() as conn: rows = await conn.fetch( "SELECT text FROM user_history " "WHERE user_id=$1 AND thread_id=$2 " "ORDER BY timestamp DESC LIMIT $3", user_id, thread_id, k, ) lines = [f"Past: {r['text']}" for r in reversed(rows)] full_context = "\n".join(lines) # Truncate by approximate word count rather than raw characters. words = full_context.split() if len(words) > self._config.MAX_CONTEXT_WORDS: full_context = " ".join(words[-self._config.MAX_CONTEXT_WORDS:]) return full_context async def get_relevant_history(self, user_id: str, query: str, k: int = 6) -> str: q_emb = await self.get_embedding(query) async with self._pool.acquire() as conn: rows = await conn.fetch( "SELECT text, sentiment FROM user_history " "WHERE user_id=$1 " "ORDER BY embedding <=> $2::vector LIMIT $3", user_id, q_emb, k, ) return ( "\n".join([f"Past: {r['text']} (sentiment {r['sentiment']:.2f})" for r in rows]) or "" ) # ── Tone adjustment ─────────────────────────────────────────────────────── def adjust_tone(self, response: str, mood: str) -> str: """ Prepend a mood-appropriate prefix. Idempotent: will not prepend if the prefix is already present. """ prefix = _TONE_PREFIXES.get(mood, _TONE_PREFIXES["neutral"]) if response.startswith(prefix): return response return prefix + response
