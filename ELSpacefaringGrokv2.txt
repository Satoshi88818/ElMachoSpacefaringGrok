#!/usr/bin/env python3
import os
import json
import time
import uuid
import random
import asyncio
import logging
from typing import List, Dict, Any
from datetime import datetime
from contextlib import asynccontextmanager

import aiohttp
from fastapi import FastAPI, Depends, HTTPException, status, Request
from fastapi.security import OAuth2PasswordBearer
from pydantic import BaseModel
from jose import JWTError, jwt
import torch
import torch.nn as nn
from transformers import AutoTokenizer, AutoModel
from cachetools import TTLCache
from tenacity import retry, stop_after_attempt, wait_exponential
from sqlalchemy import String, Float, DateTime, text
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column
from sqlalchemy.ext.asyncio import create_async_engine, async_sessionmaker, AsyncSession
from prometheus_client import start_http_server
from sgp4.api import Satrec, jday

# Rate limiting
from slowapi import Limiter, _rate_limit_exceeded_handler
from slowapi.util import get_remote_address
from slowapi.errors import RateLimitExceeded

from config import config

logging.basicConfig(level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s")
logger = logging.getLogger(__name__)

# ====================== DB ======================
engine = create_async_engine(config.DATABASE_URL, echo=False, future=True)
AsyncSessionLocal = async_sessionmaker(engine, class_=AsyncSession, expire_on_commit=False)

class Base(DeclarativeBase):
    pass

class UserHistory(Base):
    __tablename__ = "user_history"
    id: Mapped[int] = mapped_column(primary_key=True, autoincrement=True)
    user_id: Mapped[str] = mapped_column(String(50), nullable=False)
    text: Mapped[str] = mapped_column(String, nullable=False)
    sentiment: Mapped[float] = mapped_column(Float, nullable=False)
    timestamp: Mapped[datetime] = mapped_column(DateTime(timezone=True), server_default=text("NOW() AT TIME ZONE 'UTC'"))

# ====================== JWT ======================
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")

def verify_token(token: str = Depends(oauth2_scheme)) -> Dict[str, str]:
    try:
        payload = jwt.decode(token, config.JWT_SECRET.get_secret_value(), algorithms=["HS256"])
        user_id: str = payload.get("sub")
        if not user_id:
            raise HTTPException(status_code=401, detail="Invalid token")
        return {"user_id": user_id}
    except JWTError:
        raise HTTPException(status_code=401, detail="Token verification failed")

# ====================== NEURAL ======================
class SpaceNeuralNetwork(nn.Module):
    def __init__(self):
        super().__init__()
        self.transformer = AutoModel.from_pretrained(config.TRANSFORMER_MODEL)
        self.dropout = nn.Dropout(config.DROPOUT_RATE)
        self.backbone = nn.Sequential(nn.Linear(self.transformer.config.hidden_size, 512), nn.ReLU())
        self.sentiment_head = nn.Linear(512, 1)
        self.intent_head = nn.Linear(512, 5)
        self.predict_head = nn.Linear(512, 6)
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.to(self.device)
        self.tokenizer = AutoTokenizer.from_pretrained(config.TRANSFORMER_MODEL)
        self.model_path = config.MODEL_PATH

    def forward(self, input_ids: torch.Tensor, attention_mask: torch.Tensor) -> Dict[str, torch.Tensor]:
        with torch.no_grad():
            transformer_out = self.transformer(input_ids=input_ids.to(self.device), attention_mask=attention_mask.to(self.device)).last_hidden_state[:, 0, :]
        features = self.backbone(self.dropout(transformer_out))
        return {
            "sentiment": torch.tanh(self.sentiment_head(features)),
            "intent": self.intent_head(features),
            "predict": self.predict_head(features)
        }

    def load(self):
        if os.path.exists(self.model_path):
            state = torch.load(self.model_path, map_location=self.device, weights_only=True)
            self.load_state_dict(state, strict=False)
            self.eval()
            logger.info(f"Loaded persisted model from {self.model_path}")
        else:
            logger.warning(f"No model found at {self.model_path}. Using base weights.")

# ====================== EMPATHY ======================
class EmpathyModule:
    def __init__(self, neural_backend: SpaceNeuralNetwork, sessionmaker, config):
        self.neural_backend = neural_backend
        self.tokenizer = neural_backend.tokenizer
        self.sessionmaker = sessionmaker
        self.config = config
        self.intents = ["command", "seek_help", "predict", "seek_insight", "anomaly"]

    async def detect_intent(self, input_text: str, user_id: str) -> Dict[str, Any]:
        try:
            inputs = self.tokenizer(input_text, return_tensors="pt", truncation=True, padding=True, max_length=128)
            outputs = self.neural_backend(inputs["input_ids"], inputs["attention_mask"])
            intent_idx = outputs["intent"].argmax().item()
            sentiment = outputs["sentiment"].item()
            confidence = self.config.DEFAULT_CONFIDENCE

            async with self.sessionmaker() as session:
                await self._store_history(session, user_id, input_text, sentiment)
                avg_history_sentiment = await self._get_avg_sentiment(session, user_id)

            combined_sentiment = 0.6 * sentiment + 0.4 * avg_history_sentiment
            mood = "stressed" if combined_sentiment < 0 else "neutral"
            return {"mood": mood, "intent": self.intents[intent_idx], "sentiment": combined_sentiment, "confidence": confidence}
        except Exception as e:
            logger.error(f"Intent detection failed: {e}")
            return {"mood": "neutral", "intent": "seek_insight", "sentiment": 0.0, "confidence": 0.5}

    async def _store_history(self, session: AsyncSession, user_id: str, text: str, sentiment: float):
        await session.execute(
            text("""
                INSERT INTO user_history (user_id, text, sentiment, timestamp)
                VALUES (:uid, :txt, :sent, NOW() AT TIME ZONE 'UTC')
                ON CONFLICT DO NOTHING
            """),
            {"uid": user_id, "txt": text[:config.MAX_QUERY_LENGTH], "sent": sentiment}
        )
        await session.execute(
            text("DELETE FROM user_history WHERE user_id = :uid AND timestamp < NOW() AT TIME ZONE 'UTC' - INTERVAL ':ttl seconds'"),
            {"uid": user_id, "ttl": self.config.HISTORY_TTL_SECONDS}
        )
        await session.commit()

    async def _get_avg_sentiment(self, session: AsyncSession, user_id: str) -> float:
        result = await session.execute(
            text("""
                SELECT AVG(sentiment) FROM user_history 
                WHERE user_id = :uid AND timestamp > NOW() AT TIME ZONE 'UTC' - INTERVAL ':ttl seconds'
            """),
            {"uid": user_id, "ttl": self.config.HISTORY_TTL_SECONDS}
        )
        avg = result.scalar_one_or_none()
        return float(avg) if avg is not None else 0.0

    def adjust_tone(self, response: str, intent_data: Dict[str, Any]) -> str:
        if intent_data["mood"] == "stressed":
            return f"Take a breath—{response}"
        return f"Locked on: {response}"

# ====================== TOOLS ======================
class HyperIntegratedTools:
    def __init__(self, config):
        self.cache = TTLCache(maxsize=1000, ttl=config.CACHE_TTL)
        self.config = config
        self.custom_dataset = self._load_custom_dataset()

    def _load_custom_dataset(self) -> Dict[str, Any]:
        try:
            with open(self.config.CUSTOM_DATASET_PATH, "r") as f:
                return json.load(f)
        except Exception:
            return {"exoplanets": [], "asteroids": [], "anomalies": []}

    async def fetch_real_time_insights(self, query: str) -> List[str]:
        if query in self.cache:
            return self.cache[query]
        async with aiohttp.ClientSession() as session:
            tasks = [
                self._fetch_iss_position(session),
                self._fetch_space_weather(session),
                self._fetch_custom_insight(query),
            ]
            results = await asyncio.gather(*tasks, return_exceptions=True)
            insights = [r for r in results if isinstance(r, str) and r]
            self.cache[query] = insights if insights else ["No cosmic signals yet."]
            return self.cache[query]

    @retry(stop=stop_after_attempt(config.API_RETRIES), wait=wait_exponential(multiplier=1, min=1, max=10))
    async def _fetch_iss_position(self, session):
        async with session.get("http://api.open-notify.org/iss-now.json", timeout=2.0) as resp:
            data = await resp.json()
            return f"ISS at Lat: {data['iss_position']['latitude']}, Lon: {data['iss_position']['longitude']}"

    async def _fetch_space_weather(self, session):
        url = f"https://api.nasa.gov/DONKI/notifications?type=all&api_key={self.config.NASA_API_KEY}"
        async with session.get(url, timeout=2.0) as resp:
            data = await resp.json()
            return f"Space weather: {data[0]['messageBody'][:50]}..." if data else "No alerts."

    async def _fetch_custom_insight(self, query: str) -> str:
        query_lower = query.lower()
        if "exoplanet" in query_lower and self.custom_dataset["exoplanets"]:
            planet = random.choice(self.custom_dataset["exoplanets"])
            return f"Exoplanet: {planet['name']} at {planet['distance']} ly."
        return ""

    def blend_context(self, response: str, insights: List[str]) -> str:
        if not insights or random.random() < 0.6:
            return response
        return f"{response} [{' | '.join(insights)}]"

# ====================== REASONING ======================
class ReasoningEngine:
    def __init__(self, max_depth: int = 10, timeout: float = 3.0):
        self.max_depth = max_depth
        self.timeout = timeout

    def analyze(self, query: str, depth: int, context: Dict[str, Any]) -> Dict[str, Any]:
        start_time = time.time()
        metrics = {"time_spent": 0.0, "confidence": 0.85}
        if time.time() - start_time > self.timeout:
            return {"insights": "Overclocked—let’s reel it back!", "metrics": metrics}
        query_lower = query.lower()
        if "predict" in query_lower:
            insights = "Prediction in progress..."
            metrics["confidence"] = 0.9
        else:
            insights = "No space signal—zero in on something cosmic!"
            metrics["confidence"] = 0.7
        metrics["time_spent"] = time.time() - start_time
        return {"insights": insights, "metrics": metrics}

# ====================== COMMAND (FULLY ASYNC) ======================
class CommandModule:
    async def plot_orbit(self, satellite: str = "ISS") -> str:
        try:
            async with aiohttp.ClientSession() as session:
                async with session.get(
                    "https://celestrak.org/NORAD/elements/gp.php?GROUP=active&FORMAT=tle",
                    timeout=5.0
                ) as resp:
                    raw = await resp.text()
            lines = raw.splitlines()
            tle1 = tle2 = None
            i = 0
            while i < len(lines) - 2:
                if lines[i].strip().upper() == satellite.upper() or \
                   (satellite.isdigit() and satellite in lines[i + 1]):
                    tle1 = lines[i + 1]
                    tle2 = lines[i + 2]
                    break
                i += 1
            if not tle1 or not tle2:
                return f"Satellite '{satellite}' not found in active catalog. Try 'ISS' or a NORAD ID (e.g. 25544)."
            sat = Satrec.twoline2rv(tle1.strip(), tle2.strip())
            now = datetime.utcnow()
            jd, fr = jday(now.year, now.month, now.day, now.hour, now.minute, now.second)
            e, r, _ = sat.sgp4(jd, fr)
            if e == 0:
                return f"Orbit for {satellite} at {now}: X={r[0]:.1f} Y={r[1]:.1f} Z={r[2]:.1f} km"
            return "Orbit data unavailable."
        except Exception as e:
            logger.error(f"Orbit calc failed: {e}")
            return "Orbit data’s dodging us—retry."

    async def process_command(self, command: str) -> str:
        tokens = command.lower().strip().split()
        if not tokens:
            return "No command. Try 'help'."
        cmd = tokens[0]
        if cmd == "help":
            return "Commands: help, status, orbit [ISS|norad] [minutes]"
        elif cmd == "status":
            return "Gravity 1.0g, Temp 20°C. All systems nominal."
        elif cmd == "orbit":
            sat = tokens[1] if len(tokens) > 1 else "ISS"
            return await self.plot_orbit(sat)
        return "Command misfired—try 'help'."

# ====================== CORE GROK ======================
class SpacefaringGrok:
    def __init__(self, config):
        self.session_id = str(uuid.uuid4())
        self.config = config
        self.neural_backend = SpaceNeuralNetwork()
        self.neural_backend.load()
        self.empathy_engine = EmpathyModule(self.neural_backend, AsyncSessionLocal, config)
        self.toolset = HyperIntegratedTools(config)
        self.reasoning_engine = ReasoningEngine()
        self.command_module = CommandModule()

    async def process_input(self, user_id: str, user_query: str) -> Dict[str, Any]:
        start_time = time.time()
        try:
            if len(user_query) > self.config.MAX_QUERY_LENGTH:
                raise HTTPException(status_code=400, detail="Query exceeds max length")

            intent_data = await self.empathy_engine.detect_intent(user_query, user_id)
            mood, intent, confidence = intent_data["mood"], intent_data["intent"], intent_data["confidence"]

            if intent == "command":
                base_response = await self.command_module.process_command(user_query)
            elif intent == "predict":
                analysis = self.reasoning_engine.analyze(user_query, depth=5, context={})
                base_response = analysis["insights"]
                confidence = max(confidence, analysis["metrics"]["confidence"])
            else:
                base_response = "Space is calling—what’s your vector?"

            insights = await self.toolset.fetch_real_time_insights(user_query)
            response = self.toolset.blend_context(base_response, insights)
            response = self.empathy_engine.adjust_tone(response, intent_data)

            if mood != "stressed" and random.random() < self.config.HUMOR_LEVEL:
                response += " Space bends time—ready to warp?"

            metrics = {"total_time": time.time() - start_time, "confidence": confidence, "session_id": self.session_id}
            return {"response": response, "follow_ups": ["What’s next?"], "metrics": metrics}
        except Exception as e:
            logger.error(f"Processing failed: {e}")
            return {
                "response": "System’s in a spin—retry.",
                "follow_ups": ["What went wrong?"],
                "metrics": {"total_time": time.time() - start_time, "confidence": 0.5, "session_id": self.session_id}
            }

# ====================== LIFESPAN + APP ======================
@asynccontextmanager
async def lifespan(app: FastAPI):
    grok_instance = SpacefaringGrok(config)
    app.state.grok = grok_instance

    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)
        await conn.execute(text("""
            CREATE INDEX IF NOT EXISTS idx_user_id_timestamp 
            ON user_history (user_id, timestamp);
        """))
    logger.info("Async Database & tables initialized.")
    start_http_server(8001)  # Prometheus
    logger.info("SpacefaringGrok v2 ready for launch 🚀")

    yield

    await engine.dispose()
    logger.info("Shutdown complete.")

# Rate limiter (Redis-backed)
limiter = Limiter(key_func=get_remote_address, storage_uri=config.REDIS_URL)
app = FastAPI(title="SpacefaringGrok Enterprise", lifespan=lifespan)
app.state.limiter = limiter
app.add_exception_handler(RateLimitExceeded, _rate_limit_exceeded_handler)

def get_grok(request: Request) -> SpacefaringGrok:
    return request.app.state.grok

class QueryRequest(BaseModel):
    query: str

@app.post("/process", response_model=dict)
@limiter.limit(config.RATE_LIMIT)
async def process_query(
    request: QueryRequest,
    user: dict = Depends(verify_token),
    grok: SpacefaringGrok = Depends(get_grok)
):

    return result

### 5. train.py (run with `python -m spacefaring_grok.train`)
```python
#!/usr/bin/env python3
import argparse
import json
import logging
import torch
import torch.nn as nn
import torch.optim as optim
from transformers import AutoTokenizer, AutoModel
from config import config

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class SpaceNeuralNetwork(nn.Module):
    def __init__(self):
        super().__init__()
        self.transformer = AutoModel.from_pretrained(config.TRANSFORMER_MODEL)
        self.dropout = nn.Dropout(config.DROPOUT_RATE)
        self.backbone = nn.Sequential(nn.Linear(self.transformer.config.hidden_size, 512), nn.ReLU())
        self.sentiment_head = nn.Linear(512, 1)
        self.intent_head = nn.Linear(512, 5)
        self.predict_head = nn.Linear(512, 6)
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.to(self.device)
        self.tokenizer = AutoTokenizer.from_pretrained(config.TRANSFORMER_MODEL)

    def forward(self, input_ids, attention_mask):
        with torch.no_grad():
            transformer_out = self.transformer(input_ids=input_ids.to(self.device), attention_mask=attention_mask.to(self.device)).last_hidden_state[:, 0, :]
        features = self.backbone(self.dropout(transformer_out))
        return {
            "sentiment": torch.tanh(self.sentiment_head(features)),
            "intent": self.intent_head(features),
            "predict": self.predict_head(features)
        }

    def train_model(self, train_data, epochs: int = 5):
        self.train()
        optimizer = optim.Adam(self.parameters(), lr=0.001)
        inputs = [self.tokenizer(d["text"], return_tensors="pt", truncation=True, padding=True, max_length=128) for d in train_data]
        labels = torch.tensor([d["labels"] for d in train_data], dtype=torch.float32).to(self.device)

        for epoch in range(epochs):
            for i in range(0, len(inputs), config.BATCH_SIZE):
                batch = inputs[i:i + config.BATCH_SIZE]
                batch_input_ids = torch.cat([b["input_ids"] for b in batch], dim=0)
                batch_attention = torch.cat([b["attention_mask"] for b in batch], dim=0)
                batch_labels = labels[i:i + config.BATCH_SIZE]
                optimizer.zero_grad()
                outputs = self(batch_input_ids, batch_attention)
                loss = nn.MSELoss()(outputs["sentiment"], batch_labels[:, 0].unsqueeze(1)) + \
                       nn.CrossEntropyLoss()(outputs["intent"], batch_labels[:, 1].long())
                loss.backward()
                optimizer.step()
            logger.info(f"Epoch {epoch + 1}/{epochs} completed")

        torch.save(self.state_dict(), config.MODEL_PATH)
        logger.info(f"Model saved to {config.MODEL_PATH}")

def main():
    parser = argparse.ArgumentParser(description="Train SpacefaringGrok model")
    parser.add_argument("--epochs", type=int, default=5, help="Number of training epochs")
    args = parser.parse_args()

    try:
        with open(config.TRAINING_DATA_PATH, "r") as f:
            train_data = json.load(f)
        model = SpaceNeuralNetwork()
        model.train_model(train_data, epochs=args.epochs)
    except FileNotFoundError:
        logger.error(f"Training data not found: {config.TRAINING_DATA_PATH}")
        logger.error("Create space_queries.json with format [{'text': '...', 'labels': [sentiment, intent_idx]}]")

if __name__ == "__main__":
    main()
```

**How to run:**

1. `pip install -r requirements.txt`
2. Copy `.env.example` → `.env` and fill secrets
3. Train first: `python -m spacefaring_grok.train --epochs 5`
4. Launch API: `uvicorn spacefaring_grok.main:app --host 0.0.0.0 --port 8000 --reload`

Everything is now fully async, rate-limited, dependency-injected, and training is cleanly separated.
